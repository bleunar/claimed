CREATE TABLE accounts (
    id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin PRIMARY KEY,
    role ENUM('admin', 'it_head', 'it_technician', 'lab_head', 'lab_assistant') NOT NULL,
    
    name VARCHAR(255) NOT NULL,
    birth_date DATE NULL,
    gender ENUM('male', 'female', 'others') NULL,
    department_name VARCHAR(100) NULL, 

    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    password_last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status ENUM('active', 'suspended', 'deleted') DEFAULT 'active',
    last_login TIMESTAMP NULL,
    
    profile_picture VARCHAR(255) NULL,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,

    INDEX idx_department (department_name)
);

CREATE TABLE laboratories (
    id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    location VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE computer_sets (
    id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin PRIMARY KEY,
    laboratory_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
    set_name VARCHAR(50) NOT NULL,
    status ENUM('active', 'maintenance') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (laboratory_id) REFERENCES laboratories(id) ON DELETE CASCADE
);

CREATE TABLE computer_set_components (
    id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin PRIMARY KEY,
    computer_set_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    component_type ENUM('monitor', 'system_unit', 'keyboard', 'mouse', 'avr', 'web_camera', 'printer', 'headset', 'other') NOT NULL,
    is_core BOOLEAN DEFAULT TRUE,
    brand_name VARCHAR(100) NOT NULL,
    serial_number VARCHAR(100), 
    status ENUM('good', 'bad', 'maintenance', 'missing', 'disposed', 'retired') DEFAULT 'good',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (computer_set_id) REFERENCES computer_sets(id) ON DELETE SET NULL,
    INDEX idx_serial (serial_number)
);

CREATE TABLE laboratory_activity (
    id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin PRIMARY KEY,
    
    account_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    lab_target_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    set_target_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    comp_target_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,

    action_type ENUM('create', 'update', 'delete', 'status_change') NOT NULL,
    summary VARCHAR(255) NOT NULL,
    
    metadata JSON NOT NULL,

    email_notification_status ENUM('pending', 'sent', 'skipped') DEFAULT 'pending',
    email_sent_at TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (account_id) REFERENCES accounts(id) ON DELETE SET NULL,
    FOREIGN KEY (lab_target_id) REFERENCES laboratories(id) ON DELETE SET NULL,
    FOREIGN KEY (set_target_id) REFERENCES computer_sets(id) ON DELETE SET NULL,
    FOREIGN KEY (comp_target_id) REFERENCES computer_set_components(id) ON DELETE SET NULL,
    
    INDEX idx_lab_status (lab_target_id, email_notification_status),
    INDEX idx_created (created_at)
);


CREATE TABLE laboratory_issues (
    id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin PRIMARY KEY,
    
    laboratory_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    computer_set_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    component_id CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    
    flagged_by CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,
    acknowledged_by CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL, 
    resolved_by CHAR(16) CHARACTER SET ascii COLLATE ascii_bin NULL,

    title VARCHAR(150) NOT NULL,
    description TEXT NOT NULL,
    priority ENUM('low', 'medium', 'high', 'critical') DEFAULT 'medium',
    status ENUM('new', 'open', 'in_progress', 'resolved', 'wont_fix', 'closed') DEFAULT 'new',
    
    resolution_notes TEXT NULL,
    resolved_at TIMESTAMP NULL,
    
    email_notification_status ENUM('pending', 'sent') DEFAULT 'pending',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (laboratory_id) REFERENCES laboratories(id) ON DELETE SET NULL,
    FOREIGN KEY (computer_set_id) REFERENCES computer_sets(id) ON DELETE SET NULL,
    FOREIGN KEY (component_id) REFERENCES computer_set_components(id) ON DELETE SET NULL,
    FOREIGN KEY (flagged_by) REFERENCES accounts(id) ON DELETE SET NULL,
    FOREIGN KEY (acknowledged_by) REFERENCES accounts(id) ON DELETE SET NULL,
    FOREIGN KEY (resolved_by) REFERENCES accounts(id) ON DELETE SET NULL,

    INDEX idx_issue_status (laboratory_id, status),
    INDEX idx_issue_email (email_notification_status)
);
